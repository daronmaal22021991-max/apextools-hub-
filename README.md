        function updateCalculator() {
            const spendInput = document.getElementById('spending-input');
            const hoursInput = document.getElementById('hours-input');
            
            const spendUSD = parseFloat(spendInput.value);
            const hoursWeekly = parseFloat(hoursInput.value);
            
            // Display native currency USD on slider label
            document.getElementById('spending-display').innerText = `$${spendUSD}`;
            document.getElementById('hours-display').innerText = `${hoursWeekly} hrs`;

            // Production Standard: ROI benchmarks for Tier-1 markets
            const benchmarkStackCostUSD = { ai: 30, vpn: 5, fin: 0 }; 
            const avgBenchmarkStackUSD = 35; 
            const efficiencyMultiplier = 0.8; // Estimated reduction in manual hours via automation tools
            
            // CALCULATIONS (Base USD, then convert)
            
            // 1. Est. Annual Savings ($X,XXX kr/€/A$)
            // Formula: (Current Spend - Benchmark Stack) * 12 + (Manual Hours Value - Saved Hours Value)
            // Manual work hour value estimated at global $20/hr Tier-1 average
            const annualSoftwareDiff = (spendUSD - avgBenchmarkStackUSD) * 12;
            const weeklyManualValue = hoursWeekly * 20; 
            const annualManualSavings = weeklyManualValue * efficiencyMultiplier * 52;
            const totalAnnualSavingsUSD = annualSoftwareDiff + annualManualSavings;

            // 2. Time Saved / Mo (XX hrs/month)
            const monthlyManualHours = hoursWeekly * 4.3; 
            const savedHoursMo = monthlyManualHours * efficiencyMultiplier;

            // UI UPDATES (Converted)
            const symbol = appState.currencySymbol;
            const rate = appState.exchangeRate;
            const annualSavingsConverted = (totalAnnualSavingsUSD * rate);
            
            // Compact output format ($X.Xk instead of $XXX,XXX) for better UI fitting
            const formatAnnual = (val) => {
                if(val < 1) return `0 kr`; // If negative savings, display 0
                if(rate > 5) { // for Nordics SEK/NOK, large number format
                     return `${(val / 1000).toFixed(1)}k kr`; 
                }
                return `${symbol}${val.toLocaleString(undefined, {maximumFractionDigits:0})}`;
            };

            document.getElementById('savings-val').innerText = formatAnnual(annualSavingsConverted);
            document.getElementById('time-saved-val').innerText = `${savedHoursMo.toFixed(0)} hrs`;
            
            // Update currency symbols on calculator results labels (for clarity)
            document.querySelectorAll('#savings-calculator .result-label').forEach((el, i) => {
                if(i===0) {
                     el.innerText = `Est. Annual Savings (${appState.currentCurrency})`;
                }
            });
        }

        /* --- 5. HEAD-TO-HEAD COMPARISON (VS Battle) --- */
        const dockBar = document.getElementById('compare-dock-bar');
        const modal = document.getElementById('compare-modal');
        const countSpan = dockBar.querySelector('.compare-count span');
        const thumbsCtn = document.getElementById('compare-thumbs-container');
        const tableCtn = document.getElementById('compare-modal-content');

        const initComparisonLogic = () => {
            initComparisonCheckboxes();
            
            document.getElementById('clear-compare').addEventListener('click', clearComparisonState);
            document.getElementById('start-comparison').addEventListener('click', openComparison);
            document.getElementById('close-compare-modal').addEventListener('click', closeComparison);
            document.getElementById('compare-modal').addEventListener('click', (e) => {
                 if(e.target === modal) closeComparison();
            });
        };

        const initComparisonCheckboxes = () => {
            document.querySelectorAll('.compare-cb-label').forEach(label => {
                const toolId = label.dataset.toolId;
                const cb = label.querySelector('input');
                
                label.addEventListener('click', (e) => {
                    e.preventDefault(); // Stop default CB double-triggering
                    const isNowChecked = !cb.checked;
                    cb.checked = isNowChecked;
                    label.classList.toggle('checked', isNowChecked);
                    
                    toggleToolComparison(toolId, isNowChecked);
                });
            });
        };

        const toggleToolComparison = (toolId, add) => {
            const list = appState.selectedToolsForCompare;
            
            if (add) {
                if (list.length >= 3) {
                    alert("Max 3 tools allowed for Battle Mode side-by-side analysis.");
                    // Revert checkbox in grid UI
                    document.querySelector(`.compare-cb-label[data-tool-id="${toolId}"] input`).checked = false;
                    document.querySelector(`.compare-cb-label[data-tool-id="${toolId}"]`).classList.remove('checked');
                    return;
                }
                if (!list.includes(toolId)) list.push(toolId);
            } else {
                appState.selectedToolsForCompare = list.filter(id => id !== toolId);
            }
            
            updateCompareDockBar();
        };

        const updateCompareDockBar = () => {
            const count = appState.selectedToolsForCompare.length;
            countSpan.innerText = count;
            
            // Clear current thumbs
            thumbsCtn.innerHTML = '';

            if (count > 0) {
                dockBar.classList.add('show');
                
                // Production: Zero-dependency dynamic thumb injection
                appState.selectedToolsForCompare.forEach(toolId => {
                    const toolName = toolsData.find(t => t.id === toolId).name;
                    const initials = toolName.slice(0, 1).toUpperCase();
                    
                    const thumb = document.createElement('div');
                    thumb.className = 'tool-thumb';
                    thumb.innerHTML = `${initials} <button class="remove-thumb" onclick="removeComparisonTool('${toolId}')">×</button>`;
                    thumbsCtn.appendChild(thumb);
                });
                
            } else {
                dockBar.classList.remove('show');
            }
        };

        // Needs global scope for inline onclick handler above
        window.removeComparisonTool = (toolId) => {
            // Update app state
            appState.selectedToolsForCompare = appState.selectedToolsForCompare.filter(id => id !== toolId);
            
            // Update Card UI in Grid (Find label, uncheck box, remove class)
            const label = document.querySelector(`.compare-cb-label[data-tool-id="${toolId}"]`);
            if (label) {
                label.querySelector('input').checked = false;
                label.classList.remove('checked');
            }
            
            updateCompareDockBar();
            if(modal.classList.contains('show')) openComparison(); // re-render table if modal open
        };

        const clearComparisonState = () => {
             // Reset grid card UI classes
             appState.selectedToolsForCompare.forEach(toolId => {
                 const label = document.querySelector(`.compare-cb-label[data-tool-id="${toolId}"]`);
                 if(label) {
                      label.querySelector('input').checked = false;
                      label.classList.remove('checked');
                 }
             });
             appState.selectedToolsForCompare = [];
             updateCompareDockBar();
             closeComparison();
        };

        /* --- COMPARISON TABLE RENDERING (Master Output - 100% Valid HTML) --- */
        const openComparison = () => {
            if (appState.selectedToolsForCompare.length < 2) {
                alert("Please select at least 2 tools to start Battle Mode comparison.");
                return;
            }
            
            const selectedTools = appState.selectedToolsForCompare.map(id => toolsData.find(t => t.id === id));
            
            // Features / Rows definition
            const features = [
                { key: 'strength', name: 'Core Strength' },
                { key: 'bestFor', name: 'Best Use Case' },
                { key: 'price', name: `Pricing Zone (${appState.currentCurrency})` },
                { key: 'pros', name: 'Top Pros' },
                { key: 'cons', name: 'Limitations (Who should skip)' },
                { key: 'klarna', name: 'Zone Integration (EU/kr/ANZ)' },
                { key: 'rating', name: 'Tier-1 Trust Rating' }
            ];

            const tableHtml = `
                <table class="compare-table">
                    <thead>
                        <tr>
                            <th class="feature-col">Technical Spec Battle</th>
                            ${selectedTools.map(t => `<th class="tool-col-header center">${t.name}</th>`).join('')}
                        </tr>
                    </thead>
                    <tbody>
                        ${features.map(feat => `
                            <tr>
                                <td class="feature-col">${feat.name}</td>${selectedTools.map(t => `<td class="${feat.key==='price' || feat.key==='rating' ? 'center' : ''}">${renderFeatureVal(feat.key, t)}</td>`).join('')}
                            </tr>
                        `).join('')}
                         <!-- Dynamic CTA Row at Bottom of Modal -->
                         <tr>
                             <td class="feature-col" style="vertical-align: middle;">Claims Expert Deal</td>
                             ${selectedTools.map(t => `
                                 <td class="center" style="border-top: 1px solid rgba(16,185,129,0.3);">
                                    ${t.coupon ? `<button class="btn btn-primary coupon-btn btn-sm mt-3" data-url="${t.url}" data-code="${t.coupon}" style="font-size:12px; padding: 10px 16px;">Claim ${t.coupon}</button>` 
                                                 : `<a href="${t.url}" target="_blank" rel="sponsored noopener" class="btn btn-primary btn-sm" style="font-size:12px; padding: 10px 16px;">Try ${t.name}</a>`}
                                 </td>
                             `).join('')}
                         </tr>
                    </tbody>
                </table>
            `;

            tableCtn.innerHTML = tableHtml;
            modal.classList.add('show');
            applyCouponLogic(); // re-init coupon copier for modal table CTAs
        };

        const renderFeatureVal = (key, tool) => {
            const check = '<span class="table-check">✓</span>';
            const cross = '<span class="table-x">×</span>';
            
            switch(key) {
                case 'strength': return tool.catTag;
                case 'bestFor': return tool.tags.join(', ');
                case 'price': return `<strong>${formatCurrency(tool.basePrice)}</strong>${tool.period}`;
                case 'pros': return tool.pros.map(p => `• ${p}`).join('<br>');
                case 'cons': return `<span style="color:#fca5a5;font-style:italic;">skip if: ${tool.cons[0]}</span>`;
                case 'klarna': 
                    const supportedRegions = ['EU', 'NO', 'ANZ'];
                    const isZoneOk = supportedRegions.includes(appState.currentRegion);
                    if (tool.klarna && isZoneOk) return `${check} Klarna/Zone Pay`;
                    if (tool.swift && isZoneOk) return `${check} Zone Bank/Fees`;
                    return `${cross} Standard FX Only`;
                case 'rating': return `<div class="stars">${'<svg><use href="#icon-star"></use></svg>'.repeat(5)}</div><strong>${tool.rating}</strong>/5`;
                default: return '-';
            }
        };

        const closeComparison = () => {
            modal.classList.remove('show');
            tableCtn.innerHTML = ''; // prevent DOM bloating when closed
        };

        /* --- 6. EEAT metodology & FAQ Accordions --- */
        const initEEATFAQAccordions = () => {
             document.querySelectorAll('.faq-trigger').forEach(trigger => {
                 trigger.addEventListener('click', () => {
                     const item = trigger.parentElement;
                     item.classList.toggle('open');
                 });
             });
        };

        /* --- 7. EXIT INTENT LOGIC (Dwell Time booster) --- */
        const initExitIntent = () => {
            let exitTriggered = localStorage.getItem('apex_exit_intent_fired');
            if (exitTriggered) return; // Production Standard: Only fire once per session/visit type
            
            let delayedFallback = setTimeout(showExitIntentModal, 45000); // Fail-safe fallback if no mouseleave detection

            // High Precision Exit-Intent Detection
            document.addEventListener('mouseout', e => {
                if (e.relatedTarget === null && e.clientY < 10 && !exitTriggered) { // Mouse moved outside top viewport
                    showExitIntentModal();
                }
            });
            
            document.getElementById('close-exit-modal').addEventListener('click', closeExitIntent);
            document.getElementById('exit-intent-modal').addEventListener('click', e => {
                 if(e.target === document.getElementById('exit-intent-modal')) closeExitIntent();
            });

            const copyDealBtn = document.querySelector('#exit-intent-modal .copy-deal-btn');
            copyDealBtn.addEventListener('click', (e) => {
                 e.preventDefault();
                 const code = "APEX70GLOBAL";
                 const url = "https://nordvpn.com/apexglobal";
                 
                 navigator.clipboard.writeText(code);
                 
                 e.currentTarget.innerText = "✓ Discount Applied & Opening!";
                 e.currentTarget.style.backgroundColor = 'rgba(255,255,255,0.1)';
                 
                 setTimeout(() => window.open(url, '_blank'), 400);
            });
        };

        function showExitIntentModal() {
            document.getElementById('exit-intent-modal').classList.add('show');
            localStorage.setItem('apex_exit_intent_fired', 'true'); // production tracking
        }

        function closeExitIntent() {
            document.getElementById('exit-intent-modal').classList.remove('show');
        }

        /* --- 8. GDPR / CCPA COOKIE CONSENT LOGIC --- */
        const initCookieBanner = () => {
             const banner = document.getElementById('cookie-banner');
             let consentSet = localStorage.getItem('apex_cookie_consent_tier1');

             if (!consentSet) {
                  banner.classList.add('show');
             }

             document.getElementById('accept-cookies').addEventListener('click', () => {
                  localStorage.setItem('apex_cookie_consent_tier1', 'accepted');
                  banner.classList.remove('show');
             });

             document.getElementById('reject-cookies').addEventListener('click', () => {
                  // Standard compliance: still need to load functionally essential stuff, 
                  // but might block marketing/monetization scripts (not applicable in this single-file zero-dep demo).
                  localStorage.setItem('apex_cookie_consent_tier1', 'rejected');
                  banner.classList.remove('show');
             });
        };
    </script>
